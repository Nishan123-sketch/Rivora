import {getWaterLevelMinMax} from "/js/minMaxCommon.js";

var df2 = new DecimalFormat("#,##0.00");

$(document).ready(function() {
	// 시간 단위 버튼 활성화
	$('[data-type="'+timeType+'"] a').addClass('on');
	
	// 시간 단위별 달력 세팅
	if (timeType == "1M") {
		setDatetimepicker1M("sdt", "sdt");
        setDatetimepicker1M("edt", "edt");
	} else {
		setDatetimepicker10M("sdt", "sdt");
		setDatetimepicker10M("edt", "edt");
	}
	
	getMinyn(timeType);
	
	// 시간 단위 클릭 이벤트
	$(".sort a").click(function() {
		var dataTimeValue = $(this).parent(".sort").data("type");

		// 시간 단위별 달력 세팅
		if (dataTimeValue == "1M") {
			setDatetimepicker1M("sdt", "sdt");
	        setDatetimepicker1M("edt", "edt");
	        
	        // 현재 시간 10분 내림
	        const now = new Date();
	        now.setSeconds(0);
	        now.setMilliseconds(0);

	        const pad = (num) => num.toString().padStart(2, '0');

	        // edt: 현재 시간 (10분 내림)
	        const edtStr = `${now.getFullYear()}-${pad(now.getMonth() + 1)}-${pad(now.getDate())} ${pad(now.getHours())}:${pad(now.getMinutes())}`;
	        $('#edt').val(edtStr);

	        // sdt: edt의 1일 전
	        const sdtTime = new Date(now);
	        sdtTime.setHours(sdtTime.getHours() - 3);
	        const sdtStr = `${sdtTime.getFullYear()}-${pad(sdtTime.getMonth() + 1)}-${pad(sdtTime.getDate())} ${pad(sdtTime.getHours())}:${pad(sdtTime.getMinutes())}`;
	        $('#sdt').val(sdtStr);
		} else {
			setDatetimepicker10M("sdt", "sdt");
			setDatetimepicker10M("edt", "edt");
			
			// 현재 시간 10분 내림
	        const now = new Date();
	        const flooredMinutes = Math.floor(now.getMinutes() / 10) * 10;
	        now.setMinutes(flooredMinutes);
	        now.setSeconds(0);
	        now.setMilliseconds(0);

	        const pad = (num) => num.toString().padStart(2, '0');

	        // edt: 현재 시간 (10분 내림)
	        const edtStr = `${now.getFullYear()}-${pad(now.getMonth() + 1)}-${pad(now.getDate())} ${pad(now.getHours())}:${pad(now.getMinutes())}`;
	        $('#edt').val(edtStr);

	        // sdt: edt의 1일 전
	        const sdtTime = new Date(now);
	        sdtTime.setDate(sdtTime.getDate() - 1);
	        const sdtStr = `${sdtTime.getFullYear()}-${pad(sdtTime.getMonth() + 1)}-${pad(sdtTime.getDate())} ${pad(sdtTime.getHours())}:${pad(sdtTime.getMinutes())}`;
	        $('#sdt').val(sdtStr);
		}
		
		getMinyn(dataTimeValue);
	});

	// 관측소 변경 이벤트
	$("#obsnmSel").change(function(){
		getMinyn();
	});
	
}).ajaxStart(function(){
	$('#loading').show();
}).ajaxStop(function(){
	$('#loading').hide();
});

// 검색 버튼 클릭 이벤트
$('.btn.search').click(function() {
	var searchSdt = getDateTime2Blank($('#sdt').val());
	var searchEdt = getDateTime2Blank($('#edt').val());
	
	var dateStart = new Date(getDateTime2SeparatorforDate(searchSdt));
	var dateEnd = new Date(getDateTime2SeparatorforDate(searchEdt));

	// 선택된 시간 단위에 맞는 기간 제한(gap) 적용
	var gap = gapMap[timeType] || gapMap["10M"];

	// 기간 검색 제한
	if (searchDateLimit(dateStart, dateEnd, gap)) {
		if($('#obsnmSel option:selected').val() != 'NULL'){
			getMinyn(timeType);
		} else {
			popupData();
			popupChart();
		}
	}

}).ajaxStart(function(){
	$('#loading').show();
}).ajaxStop(function(){
	$('#loading').hide();
});

// 1분 단위 여부
function getMinyn(dateType) {
	if(dateType === undefined || dateType.trim() === "" || dateType == null) 	dateType = "10M";
	
	$.ajax({
		type: 'GET',
		url: '/servlet/wl/getWlMinyn.do',
		dataType: 'JSON',
		data:{
			sdt : getDateTime2Blank($('#sdt').val()),
			edt : getDateTime2Blank($('#edt').val()),
			wlobscd: $('#obsnmSel option:selected').val(),
		},
		success: function(result){
			if(result.minyn === "Y") {
				$('[data-time="1M"] a').css('display', 'none');
			}else {
				$('[data-time="1M"] a').css('display', 'block');
			}
			
			popupData();
			popupChart();
		},
		error: function(error){
		}
	});
}

// 테이블 데이터
function popupData() {
	if ($("#obsnmSel option").length == 0)
		return;
		
	$.ajax({
		type : "POST",
		url : "/servlet/wlPopupTableData.do",
		dataType : "json",
		data : {
			wlobscd : $("#obsnmSel option:selected").val(),
			sdt : getDateTime2Blank($('#sdt').val()),
			edt : getDateTime2Blank($('#edt').val()),
			type : $(".sort a.on").parent().data("type")
		},
		success : function(result){
			var html = "";

			$.each(result.tableData, function(index, value){
				var wl = isNaN(value.wl) ? "-" : df2.format(value.wl);
				var fw = isNaN(value.fw) ? "-" : df2.format(value.fw);
				var seawl = isNaN(value.seawl) ? "-" : df2.format(value.seawl);
				
				html += "<tr>";
				html += "	<td>"+ value.ymdhm +"</td>";
				html += "	<td>"+ wl +"</td>";
				html += "	<td>"+ fw +"</td>";
				html += "	<td>"+ seawl +"</td>";
				html += "</tr>";
			});
			
			$("#wltbody").empty();
			$("#wltbody").append(html);
			
			$(".popTitle").html(result.info.obsnm + " 수위 자료");
			$("#agcif").html("관할기관 : " + result.info.agcnm);
			$("#addr").html(result.info.addr + " " + result.info.etcaddr); 
		},
		error: function(textStatus, errorThrown) {
	        // 에러 처리 및 로그 출력
	        console.error('Ajax Error:', textStatus, errorThrown);
	    }
	});
}

function popupChart() {
	if ($("#obsnmSel option").length == 0)
		return;

	$.ajax({
		type : "POST",
		url : "/servlet/wlPopupChart.do",
		dataType : "json",
		data : {
			wlobscd : $("#obsnmSel option:selected").val(),
			sdt : getDateTime2Blank($('#sdt').val()),
			edt : getDateTime2Blank($('#edt').val()),
			type : $(".sort a.on").parent().data("type")
		},
		success : function(result){
			var wlobscd = result.info.wlobscd;
			var time = result.chart.timeList;
			var pmymd = result.info.pmymd;
			var pmwl = result.chart.pmwlList == null ? Array(time.length).fill(null) : result.chart.pmwlList;
			var gdt = result.info.gdt;
			var wl = result.chart.wlList == null ? Array(time.length).fill(null) : result.chart.wlList;
			var att = result.chart.attList == null ? Array(time.length).fill(null) : result.chart.attList;
			var wrn = result.chart.wrnList == null ? Array(time.length).fill(null) : result.chart.wrnList;
			var alm = result.chart.almList == null ? Array(time.length).fill(null) : result.chart.almList;
			var srs = result.chart.srsList == null ? Array(time.length).fill(null) : result.chart.srsList;
			var area = result.chart.areaList;
			
			var areaXYvar = [];
			var areaXvar = [];
			var areaYvar = [];
			var inArr = [];

			if (area != null) {
			    $.each(result.chart.areaList, function(index, value) {
			        areaXvar.push(Math.floor(value[0]));
		            areaYvar.push(Number(value[1]) - gdt);
			    });
			}

			if (areaXvar.length > 0 && areaYvar.length > 0) {
				for (var i = 0; i < areaXvar.length; i++) {
					areaXYvar.push([areaXvar[i], areaYvar[i]]);
				}
			}
			
			var inMax = areaXvar[areaXvar.length-1] / 9;
			for(var i=0; i<=9; i++){
				inArr.push(Math.floor(inMax*i));
			}

			drawChart(wlobscd, time, pmymd, pmwl, gdt, wl, att, wrn, alm, srs, areaXYvar, areaYvar, inArr, areaXvar);
		},
		error: function(textStatus, errorThrown) {
	        // 에러 처리 및 로그 출력
	        console.error('Ajax Error:', textStatus, errorThrown);
	    }
	});
}

//엑셀 다운로드
async function downloadExcel(){
	const list = extractDataFromTable('wltbody', ['ymdhm', 'wl', 'fw', 'seawl']);
	const obsnm = $("#obsnmSel option:selected").text();
	const addr = $("#addr").html();
	const ymdhm = $('#sdt').val() + "~" + $('#edt').val();
	
	//차트 이미지 Blob으로 추출하는 함수
	const getImageBlob = async (chartId) => {
		const blob = await domtoimage.toBlob(document.getElementById(chartId));
		return new Promise((resolve) => {
			const reader = new FileReader();
			reader.readAsDataURL(blob);
			reader.onloadend = () => resolve(reader.result);
		});
	};
	
	const chartImage = await getImageBlob('chart');
	
	const data = {
		list: list,
		obsnm: obsnm,
		addr: addr,
		ymdhm: ymdhm,
		img: chartImage,
	};
	
	try {
		const response = await axios({
			method: 'POST',
			url: '/servlet/wl/downloadPopupData.do',
			responseType: 'blob',
			data: data,
		});

		const url = window.URL.createObjectURL(new Blob([response.data], { type: response.headers['content-type'] }));
		const link = document.createElement('a');
		link.href = url;
		link.setAttribute('download', obsnm + ' 수위 자료.xlsx');
		document.body.appendChild(link);
		link.click();
	} catch(error) {
		console.log("Error in downloadExcel :", error);
	} finally {
	
	}
}

//테이블로부터 데이터 추출
//Params : tbodyId(String), dataFields(String Array)
function extractDataFromTable(tbodyId, dataFields) {
	const tbody = document.getElementById(tbodyId);
	const rows = tbody.querySelectorAll('tr');

	const data = [];
	rows.forEach((row) => {
		const cells = row.querySelectorAll('td');
		const rowData = {};

		cells.forEach((cell, index) => {
			rowData[dataFields[index]] = cell.textContent.trim();
		});

		data.push(rowData);
	});

	return data;
}
